# SentinelDesk — packages for the LITE image (the default).
#
# One package per line, # for comments, blank lines ignored. The Dockerfile
# feeds this straight to apt, so editing what the desktop ships means editing a
# list — no build syntax, no rebuilding a mental model of a RUN instruction
# split over forty lines.
#
# "Lite" does not mean headless. The desktop IS the product; lite means the
# smallest set that leaves nobody reaching for apt on their first afternoon.
# Anything that costs more than about 50 MB and serves a minority belongs in
# full.txt instead.
#
# Sizes below are measured, not guessed: apt-cache Installed-Size summed over
# the package and everything it drags in.

# ── Shell, in the Raspberry Pi OS style ───────────────────────────────────────
lxpanel
pcmanfm
feh
librsvg2-bin

# ── Theme, icons, fonts ───────────────────────────────────────────────────────
adwaita-icon-theme
papirus-icon-theme
fonts-roboto
fonts-noto-core

# ── Core applications ─────────────────────────────────────────────────────────
chromium
lxterminal
zenity
dbus-x11

# Media player. Doubles as a way to test the upstream audio path: VLC can
# record from the virtual microphone the browser publishes.
vlc

# ── Remote access, outbound ───────────────────────────────────────────────────
# This desktop is often the jump host, not the destination.
tigervnc-viewer
freerdp3-x11

# ── Graphics ──────────────────────────────────────────────────────────────────
libgl1-mesa-dri
mesa-vulkan-drivers
# 1.6 MB, and it answers "is acceleration actually working, or am I on
# llvmpipe?" — the first question anybody asks when the video looks slow.
vulkan-tools

# ── Desktop utilities ─────────────────────────────────────────────────────────
# A desktop with no text editor and no image viewer is a demo. These are the
# LXDE-family versions: they match the shell, start instantly, and together
# cost less than one GNOME application would.
mousepad                  # 16.8 MB · text editor with tabs and highlighting
gpicview                  #  0.6 MB · image viewer
xarchiver                 #  1.7 MB · archives, graphically
galculator                #  1.7 MB · calculator
lxtask                    #  0.4 MB · task manager

# Two PDF readers on purpose: evince is the one people expect and it copes with
# a scanned, form-bearing, badly-made file; zathura opens instantly and is what
# you want for the fifth manual of the afternoon. zathura needs the poppler
# plugin to open anything at all — without it the window comes up empty.
evince                    # 43.3 MB
zathura                   # 33.8 MB
zathura-pdf-poppler

# ── Network and diagnosis ─────────────────────────────────────────────────────
# This desktop exists to reach other machines, so the tools for finding out why
# it cannot are not an extra.
iputils-ping              #  0.2 MB
net-tools                 #  1.0 MB
nmap                      # 27.2 MB · the biggest of the group, and the one
                          #           people open the terminal for
dnsutils                  # 10.2 MB · dig, nslookup
traceroute                #  0.2 MB
mtr-tiny                  #  0.3 MB
iperf3                    #  0.4 MB · earns its place twice over on a product
                          #           whose whole pitch is throughput
tcpdump                   #  1.7 MB
socat                     #  1.8 MB
whois                     #  0.4 MB
sngrep                    #  0.3 MB · SIP call flow off the wire

# OpenVPN, console plus the openvpn-connect launcher. Needs /dev/net/tun and
# CAP_NET_ADMIN, which the compose files grant.
#
# NOT wireguard: that package is 118 MB of dracut, initramfs-tools and kmod —
# kernel-module machinery that cannot do anything inside a container.
openvpn                   #  2.1 MB

# ── Terminal ──────────────────────────────────────────────────────────────────
# The short list whose absence stops a session dead: no pager, no way to open
# an archive, no editor over SSH.
less
tree
ncdu
pv
jq
rsync
unzip
zip
p7zip-full
vim
tmux

# Listed explicitly because they used to arrive as a dependency of
# steam-installer. Removing Steam took them with it, and a desktop that cannot
# fetch a file is missing something nobody thinks to check for until they need
# it.
curl
wget

# 99.4 MB, most of it Perl. It earns that on a desktop people are meant to work
# in rather than only look at.
git


# ── From Raspberry Pi OS's own base, and worth the same here ──────────────────
# Checked against pi-gen stage2 and the rpd-* meta-packages. These are the ones
# whose absence a person notices within an hour, and together they cost about
# what one icon theme does.
bash-completion           #  1.9 MB · tab completion; the terminal stops fighting back
man-db                    #  7.8 MB · a desktop full of tools and no way to read
                          #           their manuals is half a desktop
psmisc                    #  0.9 MB · killall, pstree, fuser
usbutils                  #  0.3 MB · lsusb
strace                    #  3.0 MB · what is that process actually doing
ethtool                   #  0.8 MB · link speed, duplex, driver — the layer below ping

# ffmpeg is in Raspberry Pi's graphics set and belongs here for a sharper
# reason: this product records and streams, and ffmpeg is how you inspect,
# convert or repair what it produced without leaving the desktop.
ffmpeg                    #  3.0 MB

# Screenshots from a script. The rail and the MCP already take them
# interactively; this is the one you call from a shell loop.
scrot                     #  0.1 MB

# mDNS, so machines on the LAN answer to name.local. Worth having on a desktop
# whose job is often to reach the other machines on its network.
avahi-daemon              #  0.6 MB

# Mounting what other people hand you: NTFS disks and SMB shares.
ntfs-3g                   #  3.8 MB
cifs-utils                #  0.9 MB

# pip, because a Python tool is a common answer to "can you also…".
python3-pip               # 10.3 MB

# update-desktop-database, which builds /usr/share/applications/mimeinfo.cache.
# A Recommends of the desktop packages, so --no-install-recommends leaves it out
# and the image ships without the cache that maps MIME types to applications.
desktop-file-utils        #  0.1 MB

# ── Localisation ──────────────────────────────────────────────────────────────
# Read at start from TZ and KEYBOARD_LAYOUT, so one image serves every region.
tzdata
xkb-data
